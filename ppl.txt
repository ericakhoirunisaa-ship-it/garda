(async () => {

    // ==========================
    // KONFIGURASI
    // ==========================
    const DELAY_MIN = 1000; // 1 detik
    const DELAY_MAX = 2000; // 2 detik
    const MAX_RETRY = 5;

    const sleep = ms => new Promise(r => setTimeout(r, ms));

    const randomDelay = () =>
        Math.floor(Math.random() * (DELAY_MAX - DELAY_MIN + 1)) + DELAY_MIN;

    // ==========================
    // XSRF TOKEN
    // ==========================
    const token = decodeURIComponent(
        document.cookie
            .split("; ")
            .find(x => x.startsWith("XSRF-TOKEN="))
            .split("=")[1]
    );

    const endpoint =
        "https://fasih-sm.bps.go.id/app/api/analytic/api/v2/assignment/report-progress-by-responsibility";

    const payload = {
        surveyPeriodId: "fd68e454-ba45-4b85-8205-f3bf777ded24",
        surveyRoleId: "6d7d919a-45e5-4779-bb87-2905b49fd31a",
        size: 10,
        page: 0,
        search: "",
        target: "TARGET_ONLY",
        region: {
            region1Id: null,
            region2Id: null,
            region3Id: null,
            region4Id: null,
            region5Id: null,
            region6Id: null,
            region7Id: null,
            region8Id: null,
            region9Id: null,
            region10Id: null
        },
        regionSummaryLevel: 6
    };

    const headers = {
        "Content-Type": "application/json",
        "X-XSRF-TOKEN": token
    };

    const rows = [];

    // ==========================
    // FETCH DENGAN RETRY
    // ==========================
    async function fetchRetry(page) {

        payload.page = page;

        for (let retry = 1; retry <= MAX_RETRY; retry++) {

            try {

                const res = await fetch(endpoint, {
                    method: "POST",
                    credentials: "include",
                    headers,
                    body: JSON.stringify(payload)
                });

                if (res.status === 429) {

                    const wait = retry * 5000;

                    console.warn(
                        `Page ${page + 1} kena 429. Retry ${retry}/${MAX_RETRY} (${wait / 1000}s)`
                    );

                    await sleep(wait);

                    continue;
                }

                if (!res.ok) {

                    throw new Error(`HTTP ${res.status}`);
                }

                const json = await res.json();

                return json;

            } catch (err) {

                console.warn(
                    `Page ${page + 1} gagal (${err.message}) Retry ${retry}/${MAX_RETRY}`
                );

                await sleep(retry * 3000);
            }
        }

        console.error(`Page ${page + 1} dilewati.`);
        return null;
    }

    // ==========================
    // REQUEST PERTAMA
    // ==========================
    const first = await fetchRetry(0);

    if (!first?.data) {

        console.error("Tidak bisa mengambil data.");
        return;
    }

    const totalElements = first.data.totalElements;
    const totalPages = Math.ceil(totalElements / payload.size);

    console.log("==================================");
    console.log("Total User :", totalElements);
    console.log("Total Page :", totalPages);
    console.log("==================================");

    // ==========================
    // LOOP HALAMAN
    // ==========================
    for (let page = 0; page < totalPages; page++) {

        const json = page === 0 ? first : await fetchRetry(page);

        if (!json?.data?.content) {

            console.warn(`Page ${page + 1} kosong.`);
            continue;
        }

        for (const user of json.data.content) {

            for (const region of user.regionSummary || []) {

                const row = {
                    Email: user.email || "",
                    SLS_Code: region.regionCode || "",
                    OPEN: 0,
                    DRAFT: 0,
                    SUBMITTED_BY_PENCACAH: 0,
                    APPROVED_BY_PENGAWAS: 0,
                    REJECTED_BY_PENGAWAS: 0,
                    REVOKED_BY_PENGAWAS: 0,
                    EDITED_BY_ADMIN_KABUPATEN: 0,
                    EDITED_BY_PENGAWAS: 0
                };

                for (const status of region.statusBreakdown || []) {

                    switch (status.status) {

                        case "OPEN":
                            row.OPEN = status.count;
                            break;

                        case "DRAFT":
                            row.DRAFT = status.count;
                            break;

                        case "SUBMITTED BY Pencacah":
                            row.SUBMITTED_BY_PENCACAH = status.count;
                            break;

                        case "APPROVED BY Pengawas":
                            row.APPROVED_BY_PENGAWAS = status.count;
                            break;

                        case "REJECTED BY Pengawas":
                            row.REJECTED_BY_PENGAWAS = status.count;
                            break;

                        case "REVOKED BY Pengawas":
                            row.REVOKED_BY_PENGAWAS = status.count;
                            break;

                        case "EDITED BY Admin Kabupaten":
                            row.EDITED_BY_ADMIN_KABUPATEN = status.count;
                            break;

                        case "EDITED BY Pengawas":
                            row.EDITED_BY_PENGAWAS = status.count;
                            break;
                    }
                }

                rows.push(row);
            }
        }

        console.log(`✅ Page ${page + 1}/${totalPages}`);

        await sleep(randomDelay());
    }

    // ==========================
    // EXPORT CSV
    // ==========================
    const csvHeaders = [
        "Email",
        "SLS_Code",
        "OPEN",
        "DRAFT",
        "SUBMITTED_BY_PENCACAH",
        "APPROVED_BY_PENGAWAS",
        "REJECTED_BY_PENGAWAS",
        "REVOKED_BY_PENGAWAS",
        "EDITED_BY_ADMIN_KABUPATEN",
        "EDITED_BY_PENGAWAS"
    ];

    const csv = [
        csvHeaders.join(","),
        ...rows.map(r =>
            csvHeaders.map(h => `"${r[h] ?? ""}"`).join(",")
        )
    ].join("\n");

    const blob = new Blob([csv], {
        type: "text/csv;charset=utf-8;"
    });

    const url = URL.createObjectURL(blob);

    const a = document.createElement("a");
    a.href = url;
    a.download = "fasih_progress_all_status.csv";
    document.body.appendChild(a);
    a.click();
    a.remove();

    URL.revokeObjectURL(url);

    console.log("==================================");
    console.log("EXPORT SELESAI");
    console.log("Total Row :", rows.length);
    console.log("==================================");

})();
